    public static void Run(string dir)
    {
        Dir = dir;
        Directory.CreateDirectory(Path.Combine(dir, "files"));
        J.Append("{\n\"files\": {\n");
        bool first = true;
        foreach (var kv in MakeFiles())
        {
            File.WriteAllBytes(Path.Combine(dir, "files", kv.Key), kv.Value);
            if (!first) J.Append(",\n");
            first = false;
            J.Append(Q(kv.Key)).Append(": ");
            DumpFile(kv.Value);
        }
        J.Append("\n},\n\"api\": {\n");
        Api();
        J.Append("\n}\n}\n");
        File.WriteAllText(Path.Combine(dir, "oracle.json"), J.ToString());
    }

    // ---------------------------------------------------------------- sample bitmaps

    static Color Px(int x, int y, int w, int h, bool alpha)
    {
        int r = (x * 255) / Math.Max(1, w - 1), g = (y * 255) / Math.Max(1, h - 1), b = ((x + y) * 37) & 255;
        int a = alpha ? ((x * 3 + y * 5) % 4 == 0 ? 0 : (x * 40 + y * 30 + 30) & 255) : 255;
        if (alpha && a == 0) { r = 0; g = 0; b = 0; }
        return Color.FromArgb(a, r, g, b);
    }

    static Bitmap Sample(int w, int h, PixelFormat f, bool alpha)
    {
        var b = new Bitmap(w, h, PixelFormat.Format32bppArgb);
        for (int y = 0; y < h; y++) for (int x = 0; x < w; x++) b.SetPixel(x, y, Px(x, y, w, h, alpha));
        if (f == PixelFormat.Format32bppArgb) return b;
        var c = b.Clone(new Rectangle(0, 0, w, h), f);
        b.Dispose();
        return c;
    }

    static Bitmap Indexed(int w, int h, PixelFormat f)
    {
        var b = new Bitmap(w, h, f);
        int bits = Image.GetPixelFormatSize(f), n = 1 << bits;
        ColorPalette pal = b.Palette;
        for (int i = 0; i < pal.Entries.Length; i++)
            pal.Entries[i] = Color.FromArgb(255, (i * 67) & 255, (i * 151) & 255, (i * 29 + 40) & 255);
        b.Palette = pal;
        BitmapData d = b.LockBits(new Rectangle(0, 0, w, h), ImageLockMode.WriteOnly, f);
        var row = new byte[d.Stride];
        for (int y = 0; y < h; y++)
        {
            Array.Clear(row, 0, row.Length);
            for (int x = 0; x < w; x++)
            {
                int v = (x + y * 3) % n;
                if (bits == 8) row[x] = (byte)v;
                else if (bits == 4) row[x >> 1] |= (byte)(v << ((x & 1) == 0 ? 4 : 0));
                else row[x >> 3] |= (byte)(v << (7 - (x & 7)));
            }
            Marshal.Copy(row, 0, d.Scan0 + y * d.Stride, d.Stride);
        }
        b.UnlockBits(d);
        return b;
    }

        using (var ms = new MemoryStream()) { img.Save(ms, f); return ms.ToArray(); }

        using (var ms = new MemoryStream()) { img.Save(ms, codec, p); return ms.ToArray(); }
    }

    static uint[] s_crc;
    static uint Crc(byte[] d, int o, int n, uint c)
    {
        if (s_crc == null) { s_crc = new uint[256]; for (uint i = 0; i < 256; i++) { uint v = i; for (int k = 0; k < 8; k++) v = (v & 1) != 0 ? 0xedb88320u ^ (v >> 1) : v >> 1; s_crc[i] = v; } }
        for (int i = 0; i < n; i++) c = s_crc[(c ^ d[o + i]) & 0xff] ^ (c >> 8);
        return c;
    }
    static void Be(Stream s, uint v) { s.WriteByte((byte)(v >> 24)); s.WriteByte((byte)(v >> 16)); s.WriteByte((byte)(v >> 8)); s.WriteByte((byte)v); }
    static void Chunk(Stream s, string type, byte[] data)
    {
        Be(s, (uint)data.Length);
        var t = Encoding.ASCII.GetBytes(type);
        s.Write(t, 0, 4); s.Write(data, 0, data.Length);
        uint c = Crc(t, 0, 4, 0xffffffff); c = Crc(data, 0, data.Length, c) ^ 0xffffffff;
        Be(s, c);
    }
    static byte[] Zlib(byte[] raw)
    {
        var ms = new MemoryStream();
        ms.WriteByte(0x78); ms.WriteByte(0x9c);
        using (var z = new DeflateStream(ms, CompressionMode.Compress, true)) z.Write(raw, 0, raw.Length);
        uint a = 1, b = 0; foreach (byte x in raw) { a = (a + x) % 65521; b = (b + a) % 65521; }
        Be(ms, (b << 16) | a);
        return ms.ToArray();
    }
    // rows: already-packed scanlines WITHOUT filter bytes. Writes filter 0 except row 1 uses Sub, row 2 Paeth.
    static byte[] Png(int w, int h, int depth, int ctype, byte[][] rows, byte[] plte, byte[] trns, bool phys144, bool interlace, int[] gama)
    {
        var s = new MemoryStream();
        s.Write(new byte[] { 0x89, 0x50, 0x4e, 0x47, 13, 10, 26, 10 }, 0, 8);
        var ih = new MemoryStream(); Be(ih, (uint)w); Be(ih, (uint)h); ih.WriteByte((byte)depth); ih.WriteByte((byte)ctype); ih.WriteByte(0); ih.WriteByte(0); ih.WriteByte((byte)(interlace ? 1 : 0));
        Chunk(s, "IHDR", ih.ToArray());
        if (gama != null) { var g = new MemoryStream(); Be(g, (uint)gama[0]); Chunk(s, "gAMA", g.ToArray()); }
        if (plte != null) Chunk(s, "PLTE", plte);
        if (trns != null) Chunk(s, "tRNS", trns);
        if (phys144) { var p = new MemoryStream(); Be(p, 5669); Be(p, 5669); p.WriteByte(1); Chunk(s, "pHYs", p.ToArray()); }
        var raw = new MemoryStream();
        foreach (var r in rows) { raw.WriteByte(0); raw.Write(r, 0, r.Length); }
        Chunk(s, "IDAT", Zlib(raw.ToArray()));
        Chunk(s, "IEND", new byte[0]);
        return s.ToArray();
    }

    // Adam7: build each pass's rows from a full picture of 'channels' bytes-per-pixel (8-bit only).
    static byte[] PngInterlaced(int w, int h, int ctype, int bpp, Func<int, int, byte[]> pix)
    {
        int[] x0 = { 0, 4, 0, 2, 0, 1, 0 }, y0 = { 0, 0, 4, 0, 2, 0, 1 }, dx = { 8, 8, 4, 4, 2, 2, 1 }, dy = { 8, 8, 8, 4, 4, 2, 2 };
        var raw = new MemoryStream();
        for (int p = 0; p < 7; p++)
        {
            int pw = (w - x0[p] + dx[p] - 1) / dx[p], ph = (h - y0[p] + dy[p] - 1) / dy[p];
            if (pw <= 0 || ph <= 0) continue;
            for (int y = 0; y < ph; y++)
            {
                raw.WriteByte(0);
                for (int x = 0; x < pw; x++) { var v = pix(x0[p] + x * dx[p], y0[p] + y * dy[p]); raw.Write(v, 0, bpp); }
            }
        }
        var s = new MemoryStream();
        s.Write(new byte[] { 0x89, 0x50, 0x4e, 0x47, 13, 10, 26, 10 }, 0, 8);
        var ih = new MemoryStream(); Be(ih, (uint)w); Be(ih, (uint)h); ih.WriteByte(8); ih.WriteByte((byte)ctype); ih.WriteByte(0); ih.WriteByte(0); ih.WriteByte(1);
        Chunk(s, "IHDR", ih.ToArray());
        Chunk(s, "IDAT", Zlib(raw.ToArray()));
        Chunk(s, "IEND", new byte[0]);
        return s.ToArray();
    }

    // GIF with LZW written "uncompressed" (a clear code before the table grows), several frames.
    class GifFrame { public int L, T, W, H, Delay, Disposal, Trans = -1; public byte[] Idx; public byte[] Local; public bool Interlace; }
    static void GifLzw(Stream s, byte[] idx, int minCode)
    {
        int clear = 1 << minCode, end = clear + 1, size = minCode + 1;
        var bits = new List<byte>(); int acc = 0, n = 0;
        Action<int> put = delegate (int code) { acc |= code << n; n += size; while (n >= 8) { bits.Add((byte)acc); acc >>= 8; n -= 8; } };
        int since = 0, limit = (1 << size) - clear - 2;
        put(clear);
        foreach (byte b in idx) { if (since >= limit) { put(clear); since = 0; } put(b); since++; }
        put(end);
        if (n > 0) bits.Add((byte)acc);
        s.WriteByte((byte)minCode);
        for (int i = 0; i < bits.Count; i += 255) { int c = Math.Min(255, bits.Count - i); s.WriteByte((byte)c); for (int k = 0; k < c; k++) s.WriteByte(bits[i + k]); }
        s.WriteByte(0);
    }
    static byte[] Gif(int w, int h, byte[] global, int bg, int loop, GifFrame[] frames)
    {
        var s = new MemoryStream();
        var hdr = Encoding.ASCII.GetBytes("GIF89a"); s.Write(hdr, 0, 6);
        s.WriteByte((byte)w); s.WriteByte((byte)(w >> 8)); s.WriteByte((byte)h); s.WriteByte((byte)(h >> 8));
        int gbits = global == null ? 0 : (int)Math.Log(global.Length / 3, 2) - 1;
        s.WriteByte((byte)(global == null ? 0 : 0x80 | 0x70 | gbits)); s.WriteByte((byte)bg); s.WriteByte(0);
        if (global != null) s.Write(global, 0, global.Length);
        if (loop >= 0)
        {
            s.WriteByte(0x21); s.WriteByte(0xff); s.WriteByte(11); var ns = Encoding.ASCII.GetBytes("NETSCAPE2.0"); s.Write(ns, 0, 11);
            s.WriteByte(3); s.WriteByte(1); s.WriteByte((byte)loop); s.WriteByte((byte)(loop >> 8)); s.WriteByte(0);
        }
        foreach (var f in frames)
        {
            s.WriteByte(0x21); s.WriteByte(0xf9); s.WriteByte(4);
            s.WriteByte((byte)((f.Disposal << 2) | (f.Trans >= 0 ? 1 : 0))); s.WriteByte((byte)f.Delay); s.WriteByte((byte)(f.Delay >> 8)); s.WriteByte((byte)(f.Trans >= 0 ? f.Trans : 0)); s.WriteByte(0);
            s.WriteByte(0x2c);
            foreach (int v in new[] { f.L, f.T, f.W, f.H }) { s.WriteByte((byte)v); s.WriteByte((byte)(v >> 8)); }
            int lbits = f.Local == null ? 0 : (int)Math.Log(f.Local.Length / 3, 2) - 1;
            s.WriteByte((byte)((f.Local != null ? 0x80 | lbits : 0) | (f.Interlace ? 0x40 : 0)));
            if (f.Local != null) s.Write(f.Local, 0, f.Local.Length);
            byte[] idx = f.Idx;
            if (f.Interlace)
            {
                var o = new List<byte>();
                foreach (var pass in new[] { new[] { 0, 8 }, new[] { 4, 8 }, new[] { 2, 4 }, new[] { 1, 2 } })
                    for (int y = pass[0]; y < f.H; y += pass[1]) for (int x = 0; x < f.W; x++) o.Add(f.Idx[y * f.W + x]);
                idx = o.ToArray();
            }
            GifLzw(s, idx, Math.Max(2, f.Local != null ? lbits + 1 : gbits + 1));
        }
        s.WriteByte(0x3b);
        return s.ToArray();
    }
    static byte[] Pal(int n, int seed) { var p = new byte[n * 3]; for (int i = 0; i < n; i++) { p[i * 3] = (byte)(i * 53 + seed); p[i * 3 + 1] = (byte)(i * 97 + seed * 3); p[i * 3 + 2] = (byte)(i * 31 + 200 - seed); } return p; }

    static void Le16(Stream s, int v) { s.WriteByte((byte)v); s.WriteByte((byte)(v >> 8)); }
    static void Le32(Stream s, int v) { s.WriteByte((byte)v); s.WriteByte((byte)(v >> 8)); s.WriteByte((byte)(v >> 16)); s.WriteByte((byte)(v >> 24)); }

    // A DIB icon entry (BITMAPINFOHEADER + table + XOR + AND).
    static byte[] IcoDib(int w, int h, int bpp, bool andMaskPattern, bool zeroAlpha)
    {
        var s = new MemoryStream();
        Le32(s, 40); Le32(s, w); Le32(s, h * 2); Le16(s, 1); Le16(s, bpp); Le32(s, 0); Le32(s, 0); Le32(s, 0); Le32(s, 0); Le32(s, 0); Le32(s, 0);
        int colors = bpp <= 8 ? 1 << bpp : 0;
        for (int i = 0; i < colors; i++) { s.WriteByte((byte)(i * 41)); s.WriteByte((byte)(i * 83)); s.WriteByte((byte)(i * 17 + 100)); s.WriteByte(0); }
        int stride = ((w * bpp + 31) / 32) * 4;
        for (int y = h - 1; y >= 0; y--)
        {
            var row = new byte[stride];
            for (int x = 0; x < w; x++)
            {
                if (bpp == 32) { Color c = Px(x, y, w, h, true); row[x * 4] = c.B; row[x * 4 + 1] = c.G; row[x * 4 + 2] = c.R; row[x * 4 + 3] = zeroAlpha ? (byte)0 : c.A; }
                else if (bpp == 24) { Color c = Px(x, y, w, h, false); row[x * 3] = c.B; row[x * 3 + 1] = c.G; row[x * 3 + 2] = c.R; }
                else if (bpp == 8) row[x] = (byte)((x + y) % 256);
                else if (bpp == 4) row[x >> 1] |= (byte)(((x + y) % 16) << ((x & 1) == 0 ? 4 : 0));
                else row[x >> 3] |= (byte)(((x + y) & 1) << (7 - (x & 7)));
            }
            s.Write(row, 0, stride);
        }
        int ms = ((w + 31) / 32) * 4;
        for (int y = h - 1; y >= 0; y--)
        {
            var row = new byte[ms];
            if (andMaskPattern) for (int x = 0; x < w; x++) if (((x / 2) + y) % 3 == 0) row[x >> 3] |= (byte)(1 << (7 - (x & 7)));
            s.Write(row, 0, ms);
        }
        return s.ToArray();
    }
    static byte[] Ico(List<byte[]> entries, List<int[]> dims)
    {
        var s = new MemoryStream();
        Le16(s, 0); Le16(s, 1); Le16(s, entries.Count);
        int off = 6 + 16 * entries.Count;
        for (int i = 0; i < entries.Count; i++)
        {
            int[] d = dims[i];
            s.WriteByte((byte)(d[0] >= 256 ? 0 : d[0])); s.WriteByte((byte)(d[1] >= 256 ? 0 : d[1])); s.WriteByte((byte)(d[2] <= 8 ? (1 << d[2]) & 255 : 0)); s.WriteByte(0);
            Le16(s, 1); Le16(s, d[2]); Le32(s, entries[i].Length); Le32(s, off);
            off += entries[i].Length;
        }
        foreach (var e in entries) s.Write(e, 0, e.Length);
        return s.ToArray();
    }

    static byte[] Bmp(int w, int h, int bpp, int compression, uint[] masks, bool topDown, int headerSize, Func<int, int, uint> pix, int dpiPpm)
    {
        int stride = ((w * bpp + 31) / 32) * 4;
        int maskBytes = (compression == 3 || compression == 6) && headerSize == 40 ? (compression == 6 ? 16 : 12) : 0;
        int colors = bpp <= 8 ? 1 << bpp : 0;
        int off = 14 + headerSize + maskBytes + colors * 4;
        var s = new MemoryStream();
        s.WriteByte((byte)'B'); s.WriteByte((byte)'M'); Le32(s, off + stride * h); Le32(s, 0); Le32(s, off);
        Le32(s, headerSize); Le32(s, w); Le32(s, topDown ? -h : h); Le16(s, 1); Le16(s, bpp); Le32(s, compression); Le32(s, stride * h); Le32(s, dpiPpm); Le32(s, dpiPpm); Le32(s, 0); Le32(s, 0);
        if (headerSize > 40) { for (int i = 0; i < 4; i++) Le32(s, masks != null && i < masks.Length ? (int)masks[i] : 0); Le32(s, 0x73524742); for (int i = 0; i < (headerSize - 40 - 20) / 4; i++) Le32(s, 0); }
        else if (maskBytes > 0) for (int i = 0; i < maskBytes / 4; i++) Le32(s, (int)masks[i]);
        for (int i = 0; i < colors; i++) { s.WriteByte((byte)(i * 41)); s.WriteByte((byte)(i * 83)); s.WriteByte((byte)(i * 17 + 100)); s.WriteByte(0); }
        for (int r = 0; r < h; r++)
        {
            int y = topDown ? r : h - 1 - r;
            var row = new byte[stride];
            for (int x = 0; x < w; x++)
            {
                uint v = pix(x, y);
                if (bpp == 32) { row[x * 4] = (byte)v; row[x * 4 + 1] = (byte)(v >> 8); row[x * 4 + 2] = (byte)(v >> 16); row[x * 4 + 3] = (byte)(v >> 24); }
                else if (bpp == 24) { row[x * 3] = (byte)v; row[x * 3 + 1] = (byte)(v >> 8); row[x * 3 + 2] = (byte)(v >> 16); }
                else if (bpp == 16) { row[x * 2] = (byte)v; row[x * 2 + 1] = (byte)(v >> 8); }
                else if (bpp == 8) row[x] = (byte)v;
            }
            s.Write(row, 0, stride);
        }
        return s.ToArray();
    }

    static byte[] Rle8(int w, int h)
    {
        // Encoded runs, one per row, then a literal run, end of line; end of bitmap.
        var px = new MemoryStream();
        for (int y = 0; y < h; y++)
        {
            px.WriteByte(3); px.WriteByte((byte)(y * 7 + 1));            // run of 3
            px.WriteByte(0); px.WriteByte(4);                             // literal of 4 (even length)
            for (int k = 0; k < 4; k++) px.WriteByte((byte)(k * 30 + y));
            int rest = w - 7; if (rest > 0) { px.WriteByte((byte)rest); px.WriteByte(200); }
            px.WriteByte(0); px.WriteByte(0);                             // end of line
        }
        px.WriteByte(0); px.WriteByte(1);
        byte[] data = px.ToArray();
        int off = 14 + 40 + 256 * 4;
        var s = new MemoryStream();
        s.WriteByte((byte)'B'); s.WriteByte((byte)'M'); Le32(s, off + data.Length); Le32(s, 0); Le32(s, off);
        Le32(s, 40); Le32(s, w); Le32(s, h); Le16(s, 1); Le16(s, 8); Le32(s, 1); Le32(s, data.Length); Le32(s, 2835); Le32(s, 2835); Le32(s, 0); Le32(s, 0);
        for (int i = 0; i < 256; i++) { s.WriteByte((byte)(i * 41)); s.WriteByte((byte)(i * 83)); s.WriteByte((byte)(i * 17 + 100)); s.WriteByte(0); }
        s.Write(data, 0, data.Length);
        return s.ToArray();
    }

    static void DumpImage(Image img, bool frames)
    {
        J.Append("{");
        J.Append("\"type\":").Append(Q(img.GetType().Name));
        J.Append(",\"w\":").Append(img.Width).Append(",\"h\":").Append(img.Height);
        J.Append(",\"fmt\":").Append(Q(img.PixelFormat.ToString()));
        J.Append(",\"raw\":").Append(Q(img.RawFormat.Guid.ToString()));
        J.Append(",\"flags\":").Append(img.Flags);
        J.Append(",\"dpix\":").Append(F(img.HorizontalResolution)).Append(",\"dpiy\":").Append(F(img.VerticalResolution));
        SizeF pd = img.PhysicalDimension;
        J.Append(",\"phys\":[").Append(F(pd.Width)).Append(",").Append(F(pd.Height)).Append("]");
        J.Append(",\"dims\":[");
        Guid[] dl = img.FrameDimensionsList;
        for (int i = 0; i < dl.Length; i++) { if (i > 0) J.Append(","); J.Append(Q(dl[i].ToString())); }
        J.Append("]");
        J.Append(",\"palette\":");
        try
        {
            ColorPalette p = img.Palette;
            J.Append("{\"flags\":").Append(p.Flags).Append(",\"entries\":[");
            for (int i = 0; i < p.Entries.Length; i++) { if (i > 0) J.Append(","); J.Append(Q(((uint)p.Entries[i].ToArgb()).ToString("x8"))); }
            J.Append("]}");
        }
        catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
        J.Append(",\"props\":[");
        PropertyItem[] items = img.PropertyItems;
        for (int i = 0; i < items.Length; i++)
        {
            if (i > 0) J.Append(",");
            J.Append("{\"id\":").Append(items[i].Id).Append(",\"type\":").Append(items[i].Type).Append(",\"len\":").Append(items[i].Len)
             .Append(",\"value\":").Append(Q(items[i].Value == null ? "" : Hex(items[i].Value))).Append("}");
        }
        J.Append("]");
        var bmp = img as Bitmap;
        if (bmp != null)
        {
            J.Append(",\"argb\":").Append(Q(Lock(bmp, PixelFormat.Format32bppArgb)));
            if ((bmp.PixelFormat & PixelFormat.Indexed) != 0 || bmp.PixelFormat == PixelFormat.Format24bppRgb || bmp.PixelFormat == PixelFormat.Format48bppRgb
                || bmp.PixelFormat == PixelFormat.Format64bppArgb || bmp.PixelFormat == PixelFormat.Format16bppRgb565 || bmp.PixelFormat == PixelFormat.Format16bppRgb555
                || bmp.PixelFormat == PixelFormat.Format32bppRgb || bmp.PixelFormat == PixelFormat.Format16bppArgb1555 || bmp.PixelFormat == PixelFormat.Format16bppGrayScale)
                J.Append(",\"native\":").Append(Q(Lock(bmp, bmp.PixelFormat)));
        }
        if (frames)
        {
            foreach (Guid g in dl)
            {
                var fd = new FrameDimension(g);
                int n = img.GetFrameCount(fd);
                J.Append(",\"count_").Append(g.ToString()).Append("\":").Append(n);
                if (n > 1 && bmp != null)
                {
                    J.Append(",\"frames_").Append(g.ToString()).Append("\":[");
                    for (int i = 0; i < n; i++)
                    {
                        img.SelectActiveFrame(fd, i);
                        if (i > 0) J.Append(",");
                        J.Append("{\"w\":").Append(img.Width).Append(",\"h\":").Append(img.Height).Append(",\"fmt\":").Append(Q(img.PixelFormat.ToString()))
                         .Append(",\"dpix\":").Append(F(img.HorizontalResolution)).Append(",\"argb\":").Append(Q(Lock(bmp, PixelFormat.Format32bppArgb))).Append("}");
                    }
                    J.Append("]");
                    img.SelectActiveFrame(fd, 0);
                }
            }
        }
        J.Append("}");
    }

    static string Lock(Bitmap b, PixelFormat f)
    {
        try
        {
            BitmapData d = b.LockBits(new Rectangle(0, 0, b.Width, b.Height), ImageLockMode.ReadOnly, f);
            int rowBytes = (b.Width * Image.GetPixelFormatSize(f) + 7) / 8;
            var all = new byte[rowBytes * b.Height];
            for (int y = 0; y < b.Height; y++) Marshal.Copy(d.Scan0 + y * d.Stride, all, y * rowBytes, rowBytes);
            string s = "stride=" + d.Stride + ":" + Hex(all);
            b.UnlockBits(d);
            return s;
        }
        catch (Exception e) { return "EX " + e.GetType().Name + " " + e.Message; }
    }

    static void DumpFile(byte[] data)
    {
        J.Append("{\"image\":");
        try { using (var img = Image.FromStream(new MemoryStream(data))) DumpImage(img, true); }
        catch (Exception e) { J.Append(Q("EX " + e.GetType().Name + " " + e.Message)); }
        // Icon class view, for .ico files.
        if (data.Length > 4 && data[0] == 0 && data[1] == 0 && data[2] == 1 && data[3] == 0)
        {
            foreach (int sz in new[] { 16, 24, 32, 48, 20, 0 })
            {
                J.Append(",\"icon").Append(sz).Append("\":");
                try
                {
                    using (var ic = sz == 0 ? new Icon(new MemoryStream(data)) : new Icon(new MemoryStream(data), sz, sz))
                    using (var tb = ic.ToBitmap())
                    {
                        J.Append("{\"size\":[").Append(ic.Width).Append(",").Append(ic.Height).Append("],\"bmp\":");
                        DumpImage(tb, false);
                        var ms = new MemoryStream(); ic.Save(ms);
                        J.Append(",\"saved\":").Append(Q(Hex(ms.ToArray())));
                        J.Append("}");
                    }
                }
                catch (Exception e) { J.Append(Q("EX " + e.GetType().Name + " " + e.Message)); }
            }
        }
        J.Append("}");
    }

    static bool firstApi = true;
    static void Key(string k) { if (!firstApi) J.Append(",\n"); firstApi = false; J.Append(Q(k)).Append(": "); }

    static readonly PixelFormat[] Formats = {
        PixelFormat.Format1bppIndexed, PixelFormat.Format4bppIndexed, PixelFormat.Format8bppIndexed,
        PixelFormat.Format16bppRgb555, PixelFormat.Format16bppRgb565, PixelFormat.Format16bppArgb1555, PixelFormat.Format16bppGrayScale,
        PixelFormat.Format24bppRgb, PixelFormat.Format32bppRgb, PixelFormat.Format32bppArgb, PixelFormat.Format32bppPArgb,
        PixelFormat.Format48bppRgb, PixelFormat.Format64bppArgb, PixelFormat.Format64bppPArgb };

    static void Api()
    {
        // New bitmaps per format.
        foreach (PixelFormat pf in Formats)
        {
            Key("new_" + pf);
            try { using (var b = new Bitmap(5, 3, pf)) DumpImage(b, false); }
            catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
        }
        // SetPixel / GetPixel round trip per format.
        Color[] probe = { Color.FromArgb(128, 10, 130, 250), Color.FromArgb(255, 1, 2, 3), Color.FromArgb(0, 200, 100, 50), Color.FromArgb(77, 255, 255, 255), Color.FromArgb(255, 7, 135, 252), Color.FromArgb(1, 254, 128, 3) };
        foreach (PixelFormat pf in Formats)
        {
            Key("setget_" + pf);
            J.Append("[");
            for (int i = 0; i < probe.Length; i++)
            {
                if (i > 0) J.Append(",");
                try { using (var b = new Bitmap(2, 2, pf)) { b.SetPixel(1, 0, probe[i]); J.Append(Q(((uint)b.GetPixel(1, 0).ToArgb()).ToString("x8") + " " + Lock(b, pf))); } }
                catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
            }
            J.Append("]");
        }
        // LockBits conversions from a 32bppArgb source, a sub-rectangle.
        using (var src = Sample(6, 4, PixelFormat.Format32bppArgb, true))
        {
            Key("lock_src"); J.Append(Q(Lock(src, PixelFormat.Format32bppArgb)));
            foreach (PixelFormat pf in Formats)
            {
                Key("lock_from32_" + pf);
                try
                {
                    BitmapData d = src.LockBits(new Rectangle(1, 1, 3, 2), ImageLockMode.ReadOnly, pf);
                    int rb = (3 * Image.GetPixelFormatSize(pf) + 7) / 8;
                    var all = new byte[rb * 2];
                    for (int y = 0; y < 2; y++) Marshal.Copy(d.Scan0 + y * d.Stride, all, y * rb, rb);
                    J.Append(Q("stride=" + d.Stride + " w=" + d.Width + " h=" + d.Height + " fmt=" + d.PixelFormat + ":" + Hex(all)));
                    src.UnlockBits(d);
                }
                catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
            }
            // Write through a converted lock: 24bpp bytes into the 32bpp image.
            foreach (PixelFormat pf in new[] { PixelFormat.Format24bppRgb, PixelFormat.Format16bppRgb565, PixelFormat.Format32bppPArgb, PixelFormat.Format64bppArgb, PixelFormat.Format48bppRgb, PixelFormat.Format16bppArgb1555 })
            {
                Key("write_into32_" + pf);
                try
                {
                    using (var dst = Sample(4, 3, PixelFormat.Format32bppArgb, true))
                    {
                        BitmapData d = dst.LockBits(new Rectangle(1, 0, 2, 2), ImageLockMode.WriteOnly, pf);
                        int bpp = Image.GetPixelFormatSize(pf) / 8;
                        for (int y = 0; y < 2; y++) { var row = new byte[2 * bpp]; for (int i = 0; i < row.Length; i++) row[i] = (byte)(i * 37 + y * 101 + 13); if (pf == PixelFormat.Format64bppArgb || pf == PixelFormat.Format48bppRgb) for (int i = 1; i < row.Length; i += 2) row[i] &= 0x1f; Marshal.Copy(row, 0, d.Scan0 + y * d.Stride, row.Length); }
                        dst.UnlockBits(d);
                        J.Append(Q(Lock(dst, PixelFormat.Format32bppArgb)));
                    }
                }
                catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
            }
            // ReadWrite lock untouched round trip on PArgb (does it lose precision?)
            Key("readwrite_parg_roundtrip");
            using (var c = (Bitmap)src.Clone())
            {
                BitmapData d = c.LockBits(new Rectangle(0, 0, 6, 4), ImageLockMode.ReadWrite, PixelFormat.Format32bppPArgb);
                c.UnlockBits(d);
                J.Append(Q(Lock(c, PixelFormat.Format32bppArgb)));
            }
            // Clone to each format.
            foreach (PixelFormat pf in Formats)
            {
                Key("clone_" + pf);
                try { using (var c = src.Clone(new Rectangle(1, 0, 4, 3), pf)) DumpImage(c, false); }
                catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
            }
            foreach (RotateFlipType rf in Enum.GetValues(typeof(RotateFlipType)))
            {
                Key("rotate_" + (int)rf);
                using (var c = (Bitmap)src.Clone()) { c.RotateFlip(rf); J.Append(Q(c.Width + "x" + c.Height + " " + Lock(c, PixelFormat.Format32bppArgb))); }
            }
            Key("maketransparent_default");
            using (var c = Sample(6, 4, PixelFormat.Format24bppRgb, false)) { c.MakeTransparent(); DumpImage(c, false); }
            Key("maketransparent_color");
            using (var c = Sample(6, 4, PixelFormat.Format32bppArgb, false)) { c.MakeTransparent(Px(2, 1, 6, 4, false)); DumpImage(c, false); }
            Key("thumb");
            using (var t = src.GetThumbnailImage(3, 2, null, IntPtr.Zero)) { J.Append(Q(t.Width + "x" + t.Height + " " + t.PixelFormat + " " + t.HorizontalResolution)); }
            Key("setres");
            using (var c = (Bitmap)src.Clone()) { c.SetResolution(150.5f, 33f); J.Append(Q(c.HorizontalResolution + " " + c.VerticalResolution + " " + F(c.PhysicalDimension.Width))); }
            Key("bitmap_from_image");
            using (var c = new Bitmap(src)) DumpImage(c, false);
            Key("bitmap_24_from_image");
            using (var s24 = Sample(5, 3, PixelFormat.Format24bppRgb, false)) using (var c = new Bitmap(s24)) DumpImage(c, false);
            Key("bounds");
            { GraphicsUnit u = GraphicsUnit.Display; RectangleF r = src.GetBounds(ref u); J.Append(Q(r + " " + u)); }
        }
        // Indexed SetPixel throws?
        Key("indexed_setpixel");
        try { using (var b = new Bitmap(2, 2, PixelFormat.Format8bppIndexed)) b.SetPixel(0, 0, Color.Red); J.Append(Q("ok")); }
        catch (Exception e) { J.Append(Q("EX " + e.GetType().Name + " " + e.Message)); }
        Key("indexed_lock_to_8");
        using (var src = Sample(4, 2, PixelFormat.Format32bppArgb, false))
        {
            try { BitmapData d = src.LockBits(new Rectangle(0, 0, 4, 2), ImageLockMode.ReadOnly, PixelFormat.Format8bppIndexed); src.UnlockBits(d); J.Append(Q("ok")); }
            catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
        }
        // Encoders list.
        Key("encoders");
        J.Append("[");
        bool fe = true;
        foreach (var c in ImageCodecInfo.GetImageEncoders())
        {
            if (!fe) J.Append(","); fe = false;
            J.Append("{\"clsid\":").Append(Q(c.Clsid.ToString())).Append(",\"fmt\":").Append(Q(c.FormatID.ToString())).Append(",\"name\":").Append(Q(c.CodecName))
             .Append(",\"dll\":").Append(Q(c.DllName ?? "")).Append(",\"desc\":").Append(Q(c.FormatDescription)).Append(",\"ext\":").Append(Q(c.FilenameExtension))
             .Append(",\"mime\":").Append(Q(c.MimeType)).Append(",\"flags\":").Append((int)c.Flags).Append(",\"ver\":").Append(c.Version).Append("}");
        }
        J.Append("]");
        Key("decoders");
        J.Append("[");
        fe = true;
        foreach (var c in ImageCodecInfo.GetImageDecoders())
        {
            if (!fe) J.Append(","); fe = false;
            J.Append("{\"clsid\":").Append(Q(c.Clsid.ToString())).Append(",\"fmt\":").Append(Q(c.FormatID.ToString())).Append(",\"name\":").Append(Q(c.CodecName))
             .Append(",\"desc\":").Append(Q(c.FormatDescription)).Append(",\"ext\":").Append(Q(c.FilenameExtension))
             .Append(",\"mime\":").Append(Q(c.MimeType)).Append(",\"flags\":").Append((int)c.Flags).Append("}");
        }
        J.Append("]");
        foreach (string mime in new[] { "image/png", "image/jpeg", "image/tiff", "image/gif", "image/bmp" })
        {
            Key("params_" + mime);
            try
            {
                ImageCodecInfo codec = null; foreach (var c in ImageCodecInfo.GetImageEncoders()) if (c.MimeType == mime) codec = c;
                using (var b = new Bitmap(2, 2))
                {

                    for (int i = 0; i < ps.Param.Length; i++) { if (i > 0) J.Append(","); J.Append(Q(ps.Param[i].Encoder.Guid + " " + ps.Param[i].Type + " " + ps.Param[i].NumberOfValues)); }
                    J.Append("]");
                }
            }
            catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
        }
        // Saved-file properties: what GDI+ writes for 32bpp PNG round trip (format after reload).
        foreach (PixelFormat pf in new[] { PixelFormat.Format32bppArgb, PixelFormat.Format24bppRgb, PixelFormat.Format32bppRgb, PixelFormat.Format8bppIndexed, PixelFormat.Format16bppRgb565 })
            foreach (ImageFormat imf in new[] { ImageFormat.Png, ImageFormat.Bmp, ImageFormat.Gif, ImageFormat.Tiff, ImageFormat.Jpeg })
            {
                Key("resave_" + pf + "_" + imf);
                try
                {
                    using (var b = pf == PixelFormat.Format8bppIndexed ? Indexed(4, 3, pf) : Sample(4, 3, pf, pf == PixelFormat.Format32bppArgb))
                    using (var r = Image.FromStream(new MemoryStream(Save(b, imf))))
                        J.Append(Q(r.PixelFormat + " " + r.RawFormat.Guid + " " + r.HorizontalResolution));
                }
                catch (Exception e) { J.Append(Q("EX " + e.GetType().Name)); }
            }
        // ToBitmap of a 4bpp icon vs FromHicon.
        Key("raw_memorybmp"); using (var b = new Bitmap(2, 2)) J.Append(Q(b.RawFormat.Guid.ToString()));
        Key("flags_new32"); using (var b = new Bitmap(2, 2)) J.Append(b.Flags);
    }
